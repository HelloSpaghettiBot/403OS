cmake_minimum_required(VERSION 3.20)
project(surface_home VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
option(SURFACE_BUILD_SHELL "Build SDL3 frontend" ON)
add_library(surface_core STATIC src/model.cpp src/status.cpp src/process.cpp src/library.cpp src/progress.cpp src/control_center.cpp src/global_home.cpp)
target_include_directories(surface_core PUBLIC include)
target_link_libraries(surface_core PUBLIC Threads::Threads)
target_compile_options(surface_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(SURFACE_BUILD_SHELL)
    find_package(SDL3 3.4 REQUIRED CONFIG)
    add_executable(surface-shell src/main.cpp src/renderer.cpp)
    target_link_libraries(surface-shell PRIVATE surface_core SDL3::SDL3)
    target_compile_options(surface-shell PRIVATE -Wall -Wextra -Wpedantic -Werror)
    install(TARGETS surface-shell RUNTIME DESTINATION bin)
    install(DIRECTORY assets/ DESTINATION share/surface-home
            FILES_MATCHING PATTERN "*.sgo" PATTERN "*.metrics")
    install(FILES assets/fonts/OFL.txt DESTINATION share/licenses/surface-home-fonts)
    include(emulator.cmake)
endif()
include(CTest)
if(SURFACE_BUILD_SHELL)
    include(controller.cmake)
endif()
if(BUILD_TESTING)
    add_executable(surface-persistent-memory-tests tests/persistent_memory_tests.cpp)
    target_include_directories(surface-persistent-memory-tests PRIVATE include)
    add_test(NAME surface-persistent-memory COMMAND surface-persistent-memory-tests)
    add_executable(surface-command-json-tests tests/control_center_json_tests.cpp src/model.cpp)
    target_include_directories(surface-command-json-tests PRIVATE include)
    target_link_libraries(surface-command-json-tests PRIVATE Threads::Threads)
    target_compile_options(surface-command-json-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME surface-command-json COMMAND surface-command-json-tests)
    add_executable(surface-global-home-tests tests/global_home_tests.cpp)
    target_link_libraries(surface-global-home-tests PRIVATE surface_core)
    add_test(NAME surface-global-home COMMAND surface-global-home-tests)
    add_executable(surface-core-tests tests/core_tests.cpp)
    target_link_libraries(surface-core-tests PRIVATE surface_core)
    add_test(NAME surface-core COMMAND surface-core-tests)
    add_executable(surface-control-tests tests/control_center_tests.cpp)
    target_link_libraries(surface-control-tests PRIVATE surface_core)
    add_test(NAME surface-controls COMMAND surface-control-tests)
    add_executable(surface-status-tests tests/status_tests.cpp)
    target_link_libraries(surface-status-tests PRIVATE surface_core)
    add_test(NAME surface-status COMMAND surface-status-tests)
    # These test executables use assert() for verification, including in Release.
    foreach(test_target IN ITEMS
            surface-persistent-memory-tests
            surface-command-json-tests
            surface-global-home-tests
            surface-core-tests
            surface-control-tests
            surface-status-tests
            surface-controller-tests
            surface-controller-setup-tests)
        if(TARGET ${test_target})
            target_compile_options(${test_target} PRIVATE
                "$<$<CXX_COMPILER_ID:GNU,Clang,AppleClang>:-UNDEBUG>")
        endif()
    endforeach()
    if(SURFACE_BUILD_SHELL)
        add_test(NAME surface-input COMMAND surface-shell --self-test)
        set_tests_properties(surface-input PROPERTIES ENVIRONMENT "SDL_VIDEO_DRIVER=dummy")
    endif()
endif()
